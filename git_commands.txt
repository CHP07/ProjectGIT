# Initializes a new Git repository in the current folder.
git init 

# Renames the current branch to main.
git branch -M main   

# Stages the selected file for the next commit.
git add <file>                        

# Creates a commit containing the staged changes.
git commit -m "message"               

# Creates a new branch named feature.
git branch feature                     

# Switches the working directory to the feature branch.
git switch feature                     

# Switches the working directory back to the main branch.
git switch main                        

# Merges feature into the currently checked-out branch.
git merge feature                      

# Displays compact commit history with branch and merge relationships.
git log --oneline --graph --decorate --all  

# Shows a concise summary of modified, staged, and untracked files.
git status --short                     

# Checks whether a file or folder is ignored by Git.
git check-ignore <file>                

# Creates a new commit that reverses the latest commit.
git revert --no-edit HEAD              

# Connects the local repository to the remote GitHub repository named origin.
git remote add origin <URL>           
 
# Displays the URLs configured for remote repositories.
git remote -v                          

# Pushes main to GitHub and sets its upstream tracking branch.
git push -u origin main                

# Pushes feature to GitHub and sets its upstream tracking branch.
git push -u origin feature
